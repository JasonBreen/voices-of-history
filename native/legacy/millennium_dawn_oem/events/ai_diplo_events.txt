# ============================================================
#  EVENTO DINAMICO GENERICO
#  Il testo (titolo/descrizione/opzioni) e' preso da chiavi di
#  localizzazione che il middleware RISCRIVE a runtime, poi lancia
#  "reload localisation" e infine "event aidiplo.1" dalla console.
#  Cosi' l'IA puo' generare eventi narrativi al volo.
# ============================================================
country_event = {
	id = aidiplo.1
	title = aidiplo.1.t
	desc = aidiplo.1.d
	picture = GFX_report_event_german_reichstag_gathering
	is_triggered_only = yes

	option = {
		name = aidiplo.1.a
		# Gli effetti meccanici sono gia' stati applicati dal middleware
		# via console. Qui restano solo conferme narrative.
		ai_chance = { factor = 1 }
	}
	option = {
		name = aidiplo.1.b
		ai_chance = { factor = 0 }
	}
}

# ============================================================
#  EVENTO "AZIONE" stile focus-tree (fa scoppiare la guerra, ecc.)
#  Gli effetti dentro le opzioni sono COMPILATI AL CARICAMENTO della mod,
#  quindi declare_war_on NON da' "errors compiling" (a differenza di
#  eval_effect a runtime). Il middleware, prima di lanciarlo, salva la
#  nazione bersaglio come event_target globale 'aidiplo_tgt' e imposta la
#  variabile 'aidiplo_act' (1=guerra, 2=annessione, 3=fantoccio).
#  Viene lanciato alla nazione che COMPIE l'azione (ROOT = chi agisce).
# ============================================================
country_event = {
	id = aidiplo.2
	title = aidiplo.2.t
	desc = aidiplo.2.d
	picture = GFX_report_event_generic_army
	is_triggered_only = yes

	option = {
		name = aidiplo.2.a   # conferma -> esegue l'azione (ROOT = chi agisce)
		if = {
			limit = { check_variable = { aidiplo_act = 1 } }   # dichiara guerra
			declare_war_on = { target = event_target:aidiplo_tgt type = annex_everything }
		}
		if = {
			limit = { check_variable = { aidiplo_act = 2 } }   # annetti
			annex_country = { target = event_target:aidiplo_tgt transfer_troops = yes }
		}
		if = {
			limit = { check_variable = { aidiplo_act = 3 } }   # rendi fantoccio
			puppet = event_target:aidiplo_tgt
		}
		if = {
			limit = { check_variable = { aidiplo_act = 4 } }   # patto di non aggressione
			diplomatic_relation = { country = event_target:aidiplo_tgt relation = non_aggression_pact active = yes }
		}
		if = {
			limit = { check_variable = { aidiplo_act = 5 } }   # alleanza / fazione
			if = { limit = { is_in_faction = no } create_faction = AIDIPLO_NEW_FACTION }
			add_to_faction = event_target:aidiplo_tgt
		}
		if = {
			limit = { check_variable = { aidiplo_act = 6 } }   # garanzia
			diplomatic_relation = { country = event_target:aidiplo_tgt relation = guarantee active = yes }
		}
		if = {
			limit = { check_variable = { aidiplo_act = 7 } }   # accesso militare
			diplomatic_relation = { country = event_target:aidiplo_tgt relation = military_access active = yes }
		}
		if = {
			limit = { check_variable = { aidiplo_act = 8 } }   # fine vassallaggio (ROOT = overlord)
			end_puppet = event_target:aidiplo_tgt
		}
		if = {
			limit = { check_variable = { aidiplo_act = 9 } }   # embargo commerciale + calo opinione
			send_embargo = event_target:aidiplo_tgt
			event_target:aidiplo_tgt = { add_opinion_modifier = { target = ROOT modifier = ai_diplo_distrust } }
		}
	}
	option = {
		name = aidiplo.2.b   # ripensaci
	}
}

# ============================================================
#  PROGRESSO DEI RIVALI (bilanciamento anti-cheat)
#  Evento nascosto e ricorrente: ogni 90 giorni le grandi potenze IA
#  ricevono un progresso MODESTO e calibrato (industria + potere politico,
#  con sfumatura ideologica), cosi' il mondo resta competitivo anche se il
#  giocatore spinge molte riforme. Si ri-lancia da solo.
# ============================================================
country_event = {
	id = aidiplo.50
	hidden = yes
	is_triggered_only = yes

	immediate = {
		every_country = {
			limit = {
				is_major = yes
				is_ai = yes
				is_subject = no
			}
			add_timed_idea = { idea = ai_diplo_rival_progress days = 100 }
			add_political_power = 40
			if = { limit = { has_government = fascism }    add_war_support = 0.01 }
			if = { limit = { has_government = communism }  add_stability = 0.01 }
			if = { limit = { has_government = democratic } add_political_power = 20 }
		}
		country_event = { id = aidiplo.50 days = 90 }
	}
}

# ============================================================
#  GIORNALE TRIMESTRALE (timer): ogni ~90 giorni logga AIDIPLO_NEWS,
#  il middleware genera l'editoriale giornalistico e lo mostra.
# ============================================================
country_event = {
	id = aidiplo.51
	hidden = yes
	is_triggered_only = yes
	immediate = {
		# PRIMA di lanciare il giornale ri-esportiamo il mondo COMPLETO (tutte le ~150
		# nazioni), non solo quelle rilevanti del contesto giornaliero: l'editoriale passa
		# in rassegna il mondo intero, quindi l'IA deve avere lo stato AGGIORNATO di TUTTI.
		# Il middleware legge prima queste righe (AIDIPLO_NATION) e poi AIDIPLO_NEWS, quindi
		# quando genera l'editoriale il contesto e' gia' completo e fresco (no invenzioni).
		ai_diplo_export_world = yes
		log = "AIDIPLO_NEWS|tag=[This.GetTag]"
		country_event = { id = aidiplo.51 days = 90 }
	}
}
